import tmi from 'tmi.js';
import { storage } from '../storage';

export class TwitchChatService {
  private client: tmi.Client | null = null;
  private connectedChannels = new Set<string>();
  private chatParticipants = new Map<string, Set<string>>();

  constructor() { this.initializeClient(); }

  private initializeClient() {
    if (!process.env.TWITCH_CLIENT_ID) return;
    this.client = new tmi.Client({
      options: { debug: false },
      connection: { reconnect: true, secure: true },
      identity: { username: 'raid_train_bot', password: `oauth:${process.env.TWITCH_ACCESS_TOKEN}` }
    });
    this.setupEventHandlers();
  }

  private setupEventHandlers() {
    if (!this.client) return;
    this.client.on('connected', () => console.log('✅ Chat bot connected'));
    this.client.on('disconnected', () => console.log('❌ Chat bot disconnected'));
    this.client.on('join', async (channel, username, self) => { if (!self) await this.trackParticipant(channel.replace('#',''), username, 'join'); });
    this.client.on('message', async (channel, tags, message, self) => { if (!self) await this.trackParticipant(channel.replace('#',''), tags.username || 'unknown', 'message'); });
    this.client.on('part', async (channel, username, self) => { if (!self) await this.trackParticipant(channel.replace('#',''), username, 'part'); });
  }

  private async trackParticipant(streamUsername: string, chatUsername: string, action: 'join' | 'message' | 'part') {
    if (!this.chatParticipants.has(streamUsername)) this.chatParticipants.set(streamUsername, new Set());
    if (action !== 'part') this.chatParticipants.get(streamUsername)?.add(chatUsername);
    await storage.upsertChatParticipant({ streamUsername, chatUsername, messageCount: action === 'message' ? 1 : 0 });
  }

  async connectToStream(username: string) { if (!this.client || this.connectedChannels.has(username)) return; await this.client.join(`#${username}`); this.connectedChannels.add(username); }
  async disconnectFromStream(username: string) { if (!this.client || !this.connectedChannels.has(username)) return; await this.client.part(`#${username}`); this.connectedChannels.delete(username); this.chatParticipants.delete(username); }
  getActiveParticipants(streamUsername: string) { return Array.from(this.chatParticipants.get(streamUsername) || []); }

  async start() { if (!this.client) return false; try { await this.client.connect(); return true; } catch { return false; } }
  async stop() { if (!this.client) return; await this.client.disconnect(); this.connectedChannels.clear(); this.chatParticipants.clear(); }
  async syncWithLiveStreams(liveStreams: string[]) { for (const username of liveStreams) await this.connectToStream(username); for (const connected of this.connectedChannels) if (!liveStreams.includes(connected)) await this.disconnectFromStream(connected); }
}

export const chatService = new TwitchChatService();

export class DiscordService {
  constructor(
    private webhookUrl: string,
    private attendanceWebhookUrl?: string
  ) {}

  async testWebhook(): Promise<boolean> {
    if (!this.webhookUrl) throw new Error('Discord webhook URL not configured');
    try {
      await axios.post(this.webhookUrl, {
        content: '✅ TwitchCord webhook test successful!',
        username: 'TwitchCord',
      });
      return true;
    } catch (err) {
      console.error('Error testing Discord webhook:', err);
      return false;
    }
  }

  async sendAttendanceUpdate(payload: { username: string; displayName: string; viewerCount: number; gameName: string; chatParticipants: string[] }) {
    if (!this.attendanceWebhookUrl) return false;
    try {
      await axios.post(this.attendanceWebhookUrl, {
        username: 'TwitchCord Attendance',
        embeds: [{
          title: `Attendance: ${payload.displayName}`,
          fields: [
            { name: 'Viewers', value: String(payload.viewerCount), inline: true },
            { name: 'Game', value: payload.gameName, inline: true },
            { name: 'Active Chat Participants', value: payload.chatParticipants.join(', ') || 'None' }
          ],
          color: 5814783
        }]
      });
      return true;
    } catch (err) {
      console.error('Error sending attendance update:', err);
      return false;
    }
  }

  async sendShutdownMessage() {
    if (!this.webhookUrl) return false;
    try {
      await axios.post(this.webhookUrl, { content: '🔴 TwitchCord raid train has been manually stopped.', username: 'TwitchCord' });
      return true;
    } catch (err) {
      console.error('Error sending shutdown message:', err);
      return false;
    }
  }
}

import axios from 'axios';

export class TwitchService {
  private clientId = process.env.TWITCH_CLIENT_ID!;
  private clientSecret = process.env.TWITCH_CLIENT_SECRET!;
  private accessToken = process.env.TWITCH_ACCESS_TOKEN!;

  async getMultipleStreamsData(usernames: string[]) {
    if (!usernames.length) return [];

    const params = new URLSearchParams();
    usernames.forEach(u => params.append('user_login', u));

    try {
      const res = await axios.get(`https://api.twitch.tv/helix/streams?${params.toString()}`, {
        headers: {
          'Client-ID': this.clientId,
          Authorization: `Bearer ${this.accessToken}`
        }
      });
      return res.data.data || [];
    } catch (err) {
      console.error('Error fetching Twitch streams:', err);
      return [];
    }
  }
}

export const twitchService = new TwitchService();

server/services/scheduler.ts

import cron, { ScheduledTask } from 'node-cron';
import { storage } from '../storage';
import { twitchService } from './twitch';
import { chatService } from './chat';
import { createDiscordService } from './discord';

export class SchedulerService {
  private task: ScheduledTask | null = null;
  private attendanceTask: ScheduledTask | null = null;
  private currentTimeSlot: any = null;

  async start() {
    if (this.task) return;
    this.task = cron.schedule('* * * * *', async () => await this.checkTimeSlots());
    this.task.start();
    console.log('🕒 Scheduler started');
    await chatService.start();
  }

  async stop() {
    this.task?.stop();
    this.attendanceTask?.stop();
    this.task = null;
    this.attendanceTask = null;
    this.currentTimeSlot = null;
    await chatService.stop();
    console.log('🛑 Scheduler stopped');
  }

  private addMinutesToTime(timeStr: string, minutes: number) {
    const [h, m] = timeStr.split(':').map(Number);
    const total = h * 60 + m + minutes;
    return `${Math.floor(total/60)%24}`.padStart(2,'0')+`:${total%60}`.padStart(2,'0');
  }

  private timeToMinutes(timeStr: string) {
    const [h, m] = timeStr.split(':').map(Number);
    return h*60+m;
  }

  private isTimeInRange(current: string, start: string, end: string) {
    const c=this.timeToMinutes(current), s=this.timeToMinutes(start), e=this.timeToMinutes(end);
    return s<=e?c>=s&&c<=e:c>=s||c<=e;
  }

  private async checkTimeSlots() {
    const slots = await storage.getActiveTimeSlots();
    if (!slots.length) return;
    const now = new Date();
    const estTime = now.toLocaleString('en-US', {timeZone:'America/New_York', hour12:false}).split(', ')[1].substring(0,5);
    const day = new Date().getDay();
    let active=null;
    for(const slot of slots) if(slot.dayOfWeek===day && this.isTimeInRange(estTime,slot.startTime,this.addMinutesToTime(slot.startTime,slot.duration))) {active=slot; break;}
    if(active && (!this.currentTimeSlot || this.currentTimeSlot.id!==active.id)) {
      console.log(`🕒 Slot started: ${active.streamDisplayName}`);
      this.currentTimeSlot = active;
      this.attendanceTask?.stop();
      this.attendanceTask = cron.schedule('*/25 * * * *', async()=>await this.checkAllStreams());
      this.attendanceTask.start();
    }
  }

  async checkAllStreams() {
    const streams = await storage.getStreams();
    if(!streams.length) return;
    const usernames = streams.map(s=>s.username);
    const liveStreams = await twitchService.getMultipleStreamsData(usernames);
    const webhookSetting = await storage.getSetting('discord_webhook_url');
    const attendanceSetting = await storage.getSetting('attendance_webhook_url');
    if(!webhookSetting?.value && !attendanceSetting?.value) return;
    const discordService = createDiscordService(webhookSetting.value, attendanceSetting?.value);
    let selected=null, liveData=null;
    if(this.currentTimeSlot) {
      const scheduled = streams.find(s=>s.username===this.currentTimeSlot.streamUsername);
      if(scheduled) {
        liveData=liveStreams.find(ls=>ls.user_login.toLowerCase()===scheduled.username.toLowerCase());
        const chat=chatService.getActiveParticipants(scheduled.username);
        if(chat.length){selected=scheduled;}
      }
    }
    if(!selected){
      for(const s of streams){
        liveData=liveStreams.find(ls=>ls.user_login.toLowerCase()===s.username.toLowerCase());
        const chat=chatService.getActiveParticipants(s.username);
        if(chat.length){selected=s;break;}
      }
    }
    if(selected && liveData){
      const participants=chatService.getActiveParticipants(selected.username);
      await discordService.sendAttendanceUpdate({username:selected.username, displayName:selected.displayName, viewerCount:liveData.viewer_count, gameName:liveData.game_name, chatParticipants:participants});
    }
  }
}

export const schedulerService = new SchedulerService();

Server/storage.ts

import { PrismaClient } from '@prisma/client';

export const prisma = new PrismaClient();

export const storage = {
  // Streams
  async getStreams() {
    return await prisma.stream.findMany();
  },
  async updateStream(id: number, data: any) {
    return await prisma.stream.update({ where: { id }, data });
  },

  // Chat participants
  async upsertChatParticipant({ streamUsername, chatUsername, messageCount }: { streamUsername: string; chatUsername: string; messageCount: number }) {
    return await prisma.chatParticipant.upsert({
      where: { streamUsername_chatUsername: { streamUsername, chatUsername } },
      update: { messageCount: { increment: messageCount } },
      create: { streamUsername, chatUsername, messageCount },
    });
  },

  async getActiveTimeSlots() {
    return await prisma.timeSlot.findMany({ where: { isActive: true } });
  },

  // Settings
  async getSetting(key: string) {
    return await prisma.setting.findUnique({ where: { key } });
  },

  // Attendance
  async recordAttendanceMentions(usernames: string[], streamUsername: string, displayName: string) {
    for (const username of usernames) {
      await prisma.attendance.create({
        data: { streamUsername, displayName, username },
      });
    }
  },

  // Discord messages
  async createDiscordMessage(data: { streamUsername: string; messageContent: string; success: boolean; errorMessage: string | null }) {
    return await prisma.discordMessage.create({ data });
  },

  // Activity logs
  async createActivityLog(data: { type: string; message: string }) {
    return await prisma.activityLog.create({ data });
  },
};

import tmi from 'tmi.js';
import { storage } from '../storage';

export class TwitchChatService {
  private client: tmi.Client | null = null;
  private connectedChannels = new Set<string>();
  private chatParticipants = new Map<string, Set<string>>();

  constructor() { this.initializeClient(); }

  private initializeClient() {
    if (!process.env.TWITCH_CLIENT_ID) return console.log('⚠️ No Twitch credentials');
    this.client = new tmi.Client({
      options: { debug: false },
      connection: { reconnect: true, secure: true },
      identity: { username: 'raid_train_bot', password: `oauth:${process.env.TWITCH_ACCESS_TOKEN}` },
    });
    this.setupEventHandlers();
  }

  private setupEventHandlers() {
    if (!this.client) return;
    this.client.on('connected', () => console.log('✅ Chat bot connected'));
    this.client.on('disconnected', () => console.log('❌ Chat bot disconnected'));
    this.client.on('join', (channel, username, self) => !self && this.trackParticipant(channel.replace('#', ''), username, 'join'));
    this.client.on('part', (channel, username, self) => !self && this.trackParticipant(channel.replace('#', ''), username, 'part'));
    this.client.on('message', (channel, tags, message, self) => !self && this.trackParticipant(channel.replace('#', ''), tags.username || 'unknown', 'message'));
  }

  private async trackParticipant(streamUsername: string, chatUsername: string, action: 'join'|'message'|'part') {
    if (!this.chatParticipants.has(streamUsername)) this.chatParticipants.set(streamUsername, new Set());
    if (action !== 'part') this.chatParticipants.get(streamUsername)?.add(chatUsername);
    await storage.upsertChatParticipant({ streamUsername, chatUsername, messageCount: action==='message'?1:0 });
    console.log(`👤 ${chatUsername} ${action} in ${streamUsername}'s chat`);
  }

  async connectToStream(username: string) { if (this.client && !this.connectedChannels.has(username)) { await this.client.join(`#${username}`); this.connectedChannels.add(username); } }
  async disconnectFromStream(username: string) { if (this.client && this.connectedChannels.has(username)) { await this.client.part(`#${username}`); this.connectedChannels.delete(username); this.chatParticipants.delete(username); } }
  getActiveParticipants(streamUsername: string): string[] { return Array.from(this.chatParticipants.get(streamUsername) || []); }

  async start() { if (!this.client) return false; await this.client.connect(); const streams = await storage.getStreams(); for (const s of streams) if (s.isLive) await this.connectToStream(s.username); return true; }
  async stop() { if (!this.client) return; await this.client.disconnect(); this.connectedChannels.clear(); this.chatParticipants.clear(); }
  async syncWithLiveStreams(liveStreams: string[]) { for (const username of liveStreams) await this.connectToStream(username); for (const connected of this.connectedChannels) if (!liveStreams.includes(connected)) await this.disconnectFromStream(connected); }
}

export const chatService = new TwitchChatService();

import cron, { ScheduledTask } from 'node-cron';
import { storage } from '../storage';
import { twitchService } from './twitch';
import { createDiscordService } from './discord';
import { chatService } from './chat';

export class SchedulerService {
  private task: ScheduledTask|null = null;
  private isRunning = false;
  private attendanceTask: ScheduledTask|null = null;
  private currentTimeSlot: any = null;

  async start() {
    if (this.isRunning) return;
    this.task = cron.schedule('* * * * *', async () => { await this.checkTimeSlots(); });
    this.task.start();
    this.isRunning = true;
    await chatService.start();
    console.log('Scheduler started');
  }

  async stop() {
    this.task?.stop(); this.task=null;
    this.attendanceTask?.stop(); this.attendanceTask=null;
    this.isRunning=false;
    console.log('Scheduler stopped');
  }

  private timeToMinutes(timeStr: string) { const [h,m]=timeStr.split(':').map(Number); return h*60+m; }
  private addMinutesToTime(timeStr: string, minutes: number){ const [h,m]=timeStr.split(':').map(Number); const total=h*60+m+minutes; return `${Math.floor(total/60)%24}:${total%60}`; }
  private isTimeInRange(current:string,start:string,end:string){ const c=this.timeToMinutes(current),s=this.timeToMinutes(start),e=this.timeToMinutes(end); return s<=e?c>=s&&c<=e:c>=s||c<=e; }

  async checkTimeSlots() {
    try {
      const activeTimeSlots = await storage.getActiveTimeSlots();
      if (!activeTimeSlots.length) { this.attendanceTask?.stop(); this.attendanceTask=null; this.currentTimeSlot=null; return; }

      const now = new Date(); const currentEST = now.toLocaleTimeString('en-US',{hour12:false,timeZone:'America/New_York'}).slice(0,5); const day = now.getDay();
      let activeSlot = activeTimeSlots.find(s=>s.dayOfWeek===day && this.isTimeInRange(currentEST,s.startTime,this.addMinutesToTime(s.startTime,s.duration)));
      if (activeSlot && (!this.currentTimeSlot || this.currentTimeSlot.id!==activeSlot.id)) {
        this.currentTimeSlot=activeSlot;
        this.attendanceTask?.stop();
        this.attendanceTask=cron.schedule('*/25 * * * *', async()=>{ await this.forceAttendanceUpdate(); });
        this.attendanceTask.start();
        console.log(`Time slot started for ${activeSlot.streamDisplayName}`);
      } else if (!activeSlot) { this.attendanceTask?.stop(); this.attendanceTask=null; this.currentTimeSlot=null; console.log('Time slot ended'); }
    } catch(e){ console.error('Error checking time slots:',e);}
  }

  async forceAttendanceUpdate() {
    const streams = await storage.getStreams();
    if (!streams.length) return;
    const liveStreams = await twitchService.getMultipleStreamsData(streams.map(s=>s.username));
    const webhookSetting = await storage.getSetting('discord_webhook_url');
    const attendanceSetting = await storage.getSetting('attendance_webhook_url');
    if(!webhookSetting?.value && !attendanceSetting?.value) return;
    const discordService=createDiscordService(webhookSetting?.value||'',attendanceSetting?.value);

    // Use first live streamer with chat participants
    for (const stream of streams) {
      const liveData=liveStreams.find(ls=>ls.user_login.toLowerCase()===stream.username.toLowerCase());
      if(!liveData) continue;
      const participants=chatService.getActiveParticipants(stream.username);
      if(!participants.length) continue;
      await discordService.sendAttendanceUpdate({
        username: stream.username,
        displayName: stream.displayName,
        viewerCount: liveData.viewer_count,
        gameName: liveData.game_name,
        chatParticipants: participants
      });
      console.log(`Attendance sent for ${stream.displayName}`);
      break;
    }
  }
}

export const schedulerService = new SchedulerService();
